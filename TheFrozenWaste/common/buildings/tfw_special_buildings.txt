	type_icon = "nidavellir_icon.dds"

	# We only build this through Decision -> Great Projects
	can_construct_potential = {
		always = no # This should never show up normally
	}

	is_enabled = {
		is_dwarf = yes
	}

	show_disabled = yes

	rebuild_cost = {
		gold = {
			if = {
				limit = { is_ai = yes }
				value = ai_rebuild_value
			}
			else = { value = standard_rebuild_value }
		}
		piety = {
			if = {
				limit = { is_ai = yes }
				value = ai_rebuild_value
			}
			else = { value = standard_rebuild_value }
		}
	}

	can_rebuild = {
		custom_tooltip = {
			text = no_orcish_special_building
			NOT = { has_building_with_flag = desecrated_dwarven_capital }
		}
		is_dwarf = yes
		has_building_or_higher = dwarven_depths_01
	}

	province_modifier = {
		monthly_income = 0.5
		supply_limit_mult = 0.10
	}

	county_modifier = {
		development_growth = 0.05
		build_gold_cost = -0.05
		build_speed = -0.10
	}

	character_modifier = {
		monthly_prestige = 0.25
		monthly_crafting_lifestyle_xp_gain_mult = 0.10
	}

	fallback = {
		province_modifier = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_blue_mountains_01.dds"
	}

	desc = cleanse_angband_decision_desc
	selection_tooltip = cleanse_angband_decision_tooltip
	confirm_text = cleanse_angband_decision_confirm

	decision_group_type = realms_in_exile_major

		any_character_situation = {
			situation_type = frozen_waste
		}
		faith_is_good = yes
		NOT = { global_var:tfw_angband_cleansed = 1 }
	}

	is_valid = {
		OR = {
			custom_tooltip = {
				text = cleanse_angband_decision_tooltip_1
				any_county_in_region = {
					region = middleearth_forodwaith_angband
					holder = root
				}
			}
			custom_tooltip = {
				text = cleanse_angband_decision_tooltip_2
				any_county_in_region = {
					region = middleearth_forodwaith_angband
					count = all
					is_wasteland = yes
				}
			}
		}
		piety_level >= 4
	}

	cost = {
		piety = 2000
	}

	effect = {
		custom_tooltip = cleanse_angband_decision_effect
		set_global_variable = { name = tfw_angband_cleansed value = 1 }
		every_character_situation = {
			limit = { situation_type = frozen_waste }
			tfw_frozen_waste_monthly_tick = yes
			every_situation_participant = {
				trigger_event = tfw_situation.0001
			}
		}
		province:8105 = {
			ruin_great_building = yes
		}
		add_elven_hope_effect = { VALUE = 100 HOPE_EFFECT = effect_elven_hope_debug }
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100

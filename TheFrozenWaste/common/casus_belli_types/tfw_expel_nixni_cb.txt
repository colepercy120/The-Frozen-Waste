# Special non-conquest war to empty and ruin Angband.
# Launched by the Expel Nixni from Angband decision.
tfw_expel_nixni_angband_cb = {
    group = conquest
    combine_into_one = yes
    should_show_war_goal_subview = yes
    mutually_exclusive_titles = { always = yes }

    # Match Quest for Erebor's script-only CB pattern:
    # normal Declare War UI cannot select this CB.
    allowed_for_character = {
        always = no
    }
    allowed_for_character_display_regardless = {
        prestige_level >= 3
        piety_level >= 3
    }
    allowed_against_character = {
        scope:defender = {
            this = character:tfw_lineofnixni_001
        }
    }
    allowed_against_character_display_regardless = {}

    target_titles = all
    target_title_tier = county
    ignore_effect = change_title_holder
    valid_to_start = {
        scope:target = {
            this = title:c_thangorodrim_3
        }
        scope:defender = { is_landed = yes }
        NOT = { global_var:tfw_angband_cleansed = 1 }
    }
    should_invalidate = {
        OR = {
            NOT = { scope:defender = { is_alive = yes } }
            global_var:tfw_angband_cleansed = 1
        }
    }
    on_invalidated_desc = tfw_expel_nixni_cb_invalidated_desc
    on_invalidated = {}
    cost = {}
    on_declaration = { on_declared_war = yes }

    on_victory_desc = { desc = tfw_expel_nixni_cb_victory_desc }
    on_victory = {
        # Preserve the victorious ruler for the participant event.
        scope:attacker = {
            save_scope_as = tfw_expel_nixni_victor
            # Send before Nixni becomes landless, since she is herself a participant.
            every_character_situation = {
                limit = { situation_type = frozen_waste }
                every_situation_participant = {
                    trigger_event = tfw_situation.0009
                }
            }
        }

        # Ruin the actual Angband wonder before changing the surrounding settlements.
        province:8105 = {
            ruin_great_building = yes
        }
        title:k_angband = {
            every_de_jure_county = {
                limit = { is_wasteland = no }
                make_settlement_county_wilderness = { COUNTY = this }
            }
        }

        # Standard CK3 deposition distributes any remaining landed titles
        # (including land Nixni may have conquered outside Angband).
        # Her adventurer title is created only after she is deposed.
        scope:defender = {
            if = {
                limit = { is_landed = yes }
                depose = yes
            }
            create_landless_adventurer_title_effect = {
                REASON = flag:exile
                FLAVOR_CHAR = scope:tfw_expel_nixni_victor
            }
        }
        add_truce_attacker_victory_effect = yes
    }

    on_white_peace_desc = { desc = tfw_expel_nixni_cb_white_peace_desc }
    on_white_peace = {
        add_truce_white_peace_effect = yes
    }

    on_defeat_desc = { desc = tfw_expel_nixni_cb_defeat_desc }
    on_defeat = {
        scope:attacker = {
            add_prestige_level = -1
            add_piety_level = -1
        }
        add_truce_attacker_defeat_effect = yes
    }

    transfer_behavior = transfer
    on_primary_attacker_death = inherit
    on_primary_defender_death = inherit
    attacker_allies_inherit = yes
    defender_allies_inherit = yes

    war_name = tfw_expel_nixni_angband_war_name
    war_name_base = tfw_expel_nixni_angband_war_name
    cb_name = tfw_expel_nixni_angband_cb_name
    interface_priority = 100
    use_de_jure_wargoal_only = yes
    attacker_wargoal_percentage = 0.8
}

tfw_orc_restoration = {
	situation_group_type = major
	is_unique = yes
	gui_window_name = window_journal_entry
	illustration = "gfx/interface/illustrations/decisions/decision_elves_dwarves_05.dds"
	icon = { reference = "gfx/interface/icons/flat_icons/piety.dds" }

	sub_regions = {
		tfw_restoration_middleearth = {
			geographical_regions = { middleearth }
			map_color = { 0.3 0.7 0.45 }
		}
	}
	participant_groups = {
		good_orcs = {
			auto_add_rulers = yes
			auto_add_landless_rulers = yes
			require_realm_in_sub_region = no
			require_domicile_in_sub_region = yes
			map_color = rgb { 85 180 115 }
			icon = "gfx/interface/icons/flat_icons/piety.dds"
			is_character_valid = {
				tfw_restoration_good_orc = yes
				location = { geographical_region = middleearth }
			}
		}
	}
	on_start = {
		set_global_variable = { name = tfw_orc_restoration_started value = yes }
		set_global_variable = { name = tfw_orc_restoration_active value = yes }
		tfw_restoration_initialize_state = yes
		tfw_orc_restoration_tick = yes
	}
	on_monthly = {
		tfw_restoration_initialize_state = yes
		tfw_orc_restoration_tick = yes
	}
	on_end = {
		set_global_variable = { name = tfw_orc_restoration_active value = no }
		if = {
			limit = { has_global_variable = tfw_restoration_prayer_project }
			cancel_great_project = global_var:tfw_restoration_prayer_project
		}
	}
	start_phase = tfw_restoration_long_road
	phases = {
		tfw_restoration_long_road = {
			# No timer: only the four scripted requirements can finish this situation.
			icon = "gfx/interface/icons/flat_icons/piety.dds"
			illustration = "gfx/interface/illustrations/decisions/decision_elves_dwarves_05.dds"
			modifier_sets = {
				tfw_restoration_shared_purpose = {
					good_orcs = { character_modifier = { same_faith_opinion = 5 } }
				}
			}
		}
	}
}

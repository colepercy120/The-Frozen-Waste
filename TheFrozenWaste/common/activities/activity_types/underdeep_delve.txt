	on_start = {
		scope:host = { remove_variable = underdeep_cooldown_waived }
		# What was paid to set out, refunded if the company turns back at the entrance.
		scope:host = { save_scope_value_as = { name = ud_cost value = underdeep_activity_cost } }
		set_variable = { name = underdeep_paid value = scope:ud_cost }
		if = {
			limit = { scope:host = { is_landless_adventurer = no } }
			if = {
				limit = { has_activity_option = { category = underdeep_option_party_size option = underdeep_party_normal } }
				change_variable = { name = underdeep_paid multiply = 1.1 }
			}
			else_if = {
				limit = { has_activity_option = { category = underdeep_option_party_size option = underdeep_party_large } }
				change_variable = { name = underdeep_paid multiply = 1.25 }
			}
		}
		set_variable = { name = underdeep_gold_found value = 0 }
		set_variable = { name = underdeep_prestige_won value = 0 }
		set_variable = { name = underdeep_floor value = 1 }
		set_variable = { name = underdeep_rooms_done value = 0 }
		set_variable = { name = underdeep_boss_odds value = 0 }
		set_variable = { name = underdeep_party_deaths value = 0 }
		set_variable = { name = underdeep_chron_n value = 0 }
		activity_location = { set_variable = { name = active_underdeep_delve value = yes } }
	}

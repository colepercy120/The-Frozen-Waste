tfw_frozen_waste_monthly_tick = {
	set_variable = { name = progress_points value = 0 }
	if = {
		limit = {
			title:c_aurora_isles_2.holder = {
				faith_is_good = yes
			}
		}
		change_variable = { name = progress_points add = 25 }
	}
	if = {
		limit = {
			NOT = {
				any_county_in_region = {
					region = middleearth_forodwaith_angband
					holder = character:tfw_lineofnixni_001
				}
			}
		}
		change_variable = { name = progress_points add = 25 }
	}
	if = {
		limit = {
			global_var:tfw_angband_cleansed = 1
		}
		change_variable = { name = progress_points add = 25 }
	}
	if = {
		limit = {
			global_var:tfw_frostwood_cleansed = 1
		}
		change_variable = { name = progress_points add = 25 }
	}
	if = {
		limit = {
			var:progress_points = 100
		}
		every_situation_participant = {
			trigger_event = tfw_situation.0005
		}
		end_situation = yes
	}
}

tfw_make_halls_wastelands = {
	# If surface dwellers **somehow** manage to get underground terrain, make it wilderness ( and give them a bit of gold in return )
	if = {
		limit = {
			scope:title ?= {
				tier = tier_county
				title_province ?= { terrain = halls }
			}
			NOR = {
				is_dwarf = yes
				is_orc = yes
				is_nazgul = yes
				is_maiar_fallen = yes
				is_elf = yes
			}
		}
		make_settlement_county_wilderness = { COUNTY = scope:title }
		add_gold = 125
		debug_log = "Wilderness: Surface Race got Underground - Making Wasteland"
	}
}

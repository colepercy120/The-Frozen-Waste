### All custom invaders route through this extension hook.
### Existing upstream Eonwe and Pharazon selection is not modified.
tfw_sunset_invasion_select_custom_invader_effect = {
	if = {
		limit = { geographical_region:$GEOGRAPHICAL_REGION$ = geographical_region:tfw_santa_invasion_region }
		character:tfw_santa_001 = { save_scope_as = tfw_santa_template }
	}
}

### Use the standard Sunset Invasion dispatcher, including its spawn, landing and war hooks.
### The spawn callback replaces invader_char with the newly created Annahir before landing.
tfw_santa_invasion_effect = {
	sunset_invasion_general_effect = {
		GEOGRAPHICAL_REGION = tfw_santa_invasion_region
		GRAB_LANDING_COUNTY_EFFECT = tfw_santa_invasion_grab_landing_county
		REVIVE_DEAD_CHARACTER_EFFECT = tfw_santa_invasion_spawn_character
		INVASION_LANDING_EFFECT = tfw_santa_invasion_landing_effect
		START_INVASION_WAR_EFFECT = tfw_santa_invasion_start_war
	}
}

tfw_santa_invasion_grab_landing_county = {
	title:c_aurora_isles_2 = { save_scope_as = landing_county }
}


### Recreate Annahir using the dead historical character as his DNA template.
### The trait list and base attributes mirror the dummy, following Pharazon's revival pattern.
tfw_santa_invasion_spawn_character = {
	character:tfw_santa_001 = { save_scope_as = tfw_santa_template }
	create_character = {
		name = "Annahir"
		age = 100
		gender = male
		random_traits = no
		location = title:c_aurora_isles_2.title_province
		faith = faith:faith_santa
		culture = culture:maiar
		dynasty_house = house:house_tfw_santa
		trait = maiar
		trait = gregarious
		trait = generous
		trait = eccentric
		trait = deceitful
		trait = education_learning_5
		trait = intellect_good_3
		trait = lifestyle_mystic
		trait = schemer
		trait = scholar
		trait = twotreeslight
		trait = winter_soldier
		trait = forest_fighter
		diplomacy = 10
		martial = 9
		stewardship = 9
		intrigue = 14
		learning = 17
		prowess = 10
		after_creation = {
			### Character is in scope here (create_character documentation).
			save_scope_as = tfw_santa_spawned
			copy_inheritable_appearance_from = scope:tfw_santa_template
			### Nixni is another ancient Maia; use her current age, including TFW history corrections.
			set_age = character:tfw_lineofnixni_001.age
			set_immortal_age = 100
			add_character_flag = tfw_santa_appearance
			add_character_flag = do_not_generate_starting_family
			add_character_flag = immune_to_disease
			add_trait_xp = { trait = winter_soldier value = 100 }
			add_trait_xp = { trait = forest_fighter value = 100 }
			add_trait_xp = { trait = lifestyle_mystic value = 100 }
			### WED may have granted the historical template its Legendary Mage rank.
			if = {
				limit = { scope:tfw_santa_template = { has_trait = lifestyle_mage } }
				add_trait = lifestyle_mage
				add_trait_xp = { trait = lifestyle_mage value = 101 }
			}
		}
	}
	scope:tfw_santa_spawned = {
		give_nickname = nick_tfw_father_christmas
		save_scope_as = invader_char
	}
	### Grant titles while the created-character scope is known to exist.
	### The shared Sunset Invasion dispatcher still handles landing and war.
	if = {
		limit = { exists = scope:tfw_santa_spawned }
		tfw_santa_invasion_grant_initial_titles_effect = yes
	}
	### Make failure visible without silently handing the historical template to landing.
	if = {
		limit = {
			NOT = { exists = scope:tfw_santa_spawned }
		}
		debug_log = "TFW Sunset Invasion SPAWN FAIL: Annahir was not created."
	}
}

### Establish the invader's personal county and host title before the generic
### landing hook. This avoids depending on scope recovery after character creation.
tfw_santa_invasion_grant_initial_titles_effect = {
	if = {
		limit = {
			exists = scope:tfw_santa_spawned
			title:c_aurora_isles_2 = { NOT = { holder = scope:tfw_santa_spawned } }
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = tfw_santa_initial_county_change
			add_claim_on_loss = no
		}
		title:c_aurora_isles_2 = {
			change_title_holder = {
				holder = scope:tfw_santa_spawned
				change = scope:tfw_santa_initial_county_change
				take_baronies = yes
				### The original county holder supplies a valid landed government_base.
				government_base = scope:landing_county.holder
			}
		}
		resolve_title_and_vassal_change = scope:tfw_santa_initial_county_change
	}
	if = {
		limit = { title:c_aurora_isles_2 = { holder = scope:tfw_santa_spawned } }
		debug_log = "TFW Sunset Invasion COUNTY GRANT PASS: Annahir owns c_aurora_isles_2 after spawn."
	}
	else = {
		debug_log = "TFW Sunset Invasion COUNTY GRANT FAIL: initial title transfer did not land Annahir."
	}

	### get_title is the character effect used by the upstream landing helper.
	### Grant the Jolly Host after the county, when Annahir is landed.
	if = {
		limit = {
			title:c_aurora_isles_2 = { holder = scope:tfw_santa_spawned }
			scope:tfw_santa_spawned = { NOT = { has_title = title:h_tfw_santa_host } }
		}
		scope:tfw_santa_spawned = { get_title = title:h_tfw_santa_host }
	}
	if = {
		limit = { scope:tfw_santa_spawned = { has_title = title:h_tfw_santa_host } }
		debug_log = "TFW Sunset Invasion HOST GRANT PASS: Annahir holds h_tfw_santa_host after spawn."
	}
	else = {
		debug_log = "TFW Sunset Invasion HOST GRANT FAIL: the Jolly Host title was not assigned."
	}
}

tfw_santa_invasion_landing_effect = {
	debug_log = "TFW Sunset Invasion LANDING ENTERED: general landing callback reached."
	sunset_invasion_general_landing_effect = {
		GOVERNMENT = feudal_government
		TITLE_TO_SPAWN_WITH = title:h_tfw_santa_host
		LANDING_COUNTY_FAITH = faith:faith_santa
		LANDING_COUNTY_CULTURE = culture:lossidil
		PROVINCE_MODIFIER = tfw_santa_landing_modifier
	}

	### Assert that the new Annahir, rather than the historical dummy, owns the landing titles.
	### Recover with explicit title transfer if the shared landing helper did not grant the county.
	if = {
		limit = {
			NOT = { title:c_aurora_isles_2.holder = scope:tfw_santa_spawned }
		}
		create_title_and_vassal_change = {
			### conquest is explicitly listed among the valid transfer types in effects.log.
			type = conquest
			save_scope_as = tfw_santa_landing_change
		}
		title:c_aurora_isles_2 = {
			change_title_holder = {
				holder = scope:tfw_santa_spawned
				change = scope:tfw_santa_landing_change
				take_baronies = yes
				government_base = scope:landing_county.holder
			}
		}
		resolve_title_and_vassal_change = scope:tfw_santa_landing_change
	}
	### The hegemony is titular; grant it directly once the county has been assigned.
	if = {
		limit = { scope:tfw_santa_spawned = { NOT = { has_title = title:h_tfw_santa_host } } }
		scope:tfw_santa_spawned = { get_title = title:h_tfw_santa_host }
	}
	if = {
		limit = {
			title:c_aurora_isles_2.holder = scope:tfw_santa_spawned
			scope:tfw_santa_spawned = { has_title = title:h_tfw_santa_host }
		}
		debug_log = "TFW Sunset Invasion LANDING PASS: new Annahir holds Aurora Isles 2 and the Jolly Host."
	}
	else = {
		debug_log = "TFW Sunset Invasion LANDING FAIL: missing county or host title after explicit transfer."
	}

	scope:tfw_santa_spawned = {
		set_variable = tfw_santa_invasion_character
	}

	### Every good ruler with land in the invasion region joins Annahir's realm.
	tfw_santa_vassalize_good_rulers_effect = yes

	if = {
		limit = { NOT = { has_game_rule = no_news } }
		every_player = { trigger_event = tfw_santa_invasion.0001 }
	}
	sunset_invasion_general_variables_effect = yes
}

### Transfer good regional rulers to the Gift Lord after he gains his host title.
### Applies to independent rulers and good-aligned vassals of evil overlords.
tfw_santa_vassalize_good_rulers_effect = {
	every_ruler = {
		limit = {
			faith_is_good = yes
			is_landed = yes
			is_wastelands = no
			highest_held_title_tier < tier_hegemony
			NOT = { this = scope:invader_char }
			NOT = { is_vassal_or_below_of = scope:invader_char }
			any_sub_realm_county = { geographical_region = tfw_santa_invasion_region }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = tfw_santa_fealty_change
		}
		change_liege = {
			liege = scope:invader_char
			change = scope:tfw_santa_fealty_change
		}
		resolve_title_and_vassal_change = scope:tfw_santa_fealty_change
	}
}

### The normal event supplies an evil largest ruler; console tests may not.
### Select a different evil realm when the original target is no longer eligible.
tfw_santa_select_war_target_effect = {
	if = {
		limit = {
			exists = scope:largest_realm_ruler
			scope:largest_realm_ruler = {
				is_alive = yes
				is_independent_ruler = yes
				is_landed = yes
				faith_is_evil = yes
			}
		}
		scope:largest_realm_ruler = { save_scope_as = tfw_santa_war_target }
	}
	else = {
		ordered_independent_ruler = {
			limit = {
				is_wastelands = no
				is_landed = yes
				faith_is_evil = yes
				any_sub_realm_county = { geographical_region = tfw_santa_invasion_region }
			}
			order_by = {
				value = 0
				every_sub_realm_county = { add = 1 }
			}
			max = 1
			save_scope_as = tfw_santa_war_target
		}
	}
}

### Declare against the chosen evil ruler, matching the other Sunset Invasions.
tfw_santa_invasion_start_war = {
	tfw_santa_select_war_target_effect = yes
	if = {
		limit = { exists = scope:tfw_santa_war_target }
		### The shared defender/attacker recruitment uses largest_realm_ruler.
		scope:tfw_santa_war_target = { save_scope_as = largest_realm_ruler }
		scope:invader_char = {
			start_war = {
				cb = tfw_santa_invasion_cb
				target = scope:tfw_santa_war_target
			}
			### CK3 only places spawned regiments on the map once the invader is at war.
			spawn_army = {
				levies = 0
				men_at_arms = { type = lossidil_frostguard stacks = sunset_invasion_troop_amount }
				men_at_arms = { type = lossidil_icebowmen stacks = sunset_invasion_troop_amount }
				men_at_arms = { type = lossidil_snow_spears stacks = sunset_invasion_troop_amount }
				men_at_arms = { type = lossidil_frost_riders stacks = sunset_invasion_troop_amount }
				men_at_arms = { type = high_elven_heavy_infantry stacks = sunset_invasion_troop_amount }
				men_at_arms = { type = high_elven_archers stacks = sunset_invasion_troop_amount }
				men_at_arms = { type = song_smiths stacks = sunset_invasion_troop_amount }
				location = scope:landing_county.title_province
				origin = scope:landing_county.title_province
				inheritable = yes
				uses_supply = no
				name = tfw_santa_invasion_army_name
			}
			sunset_invasion_add_subjects_as_attackers_and_defenders_effect = {
				GEOGRAPHICAL_REGION = tfw_santa_invasion_region
			}
			create_story = story_conqueror
		}
	}
}

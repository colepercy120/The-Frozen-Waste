create_artifact_tfw_e_winter_court_crown_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = tfw_e_winter_court_crown_name
		description = tfw_e_winter_court_crown_description
		creator = character:tfw_lineofnixni_001
		type = helmet
		visuals = tfw_nixni_crown
		wealth = scope:wealth
		quality = scope:quality
		modifier = tfw_e_winter_court_crown_modifier
		decaying = no
		save_scope_as = newly_created_artifact
	}
	scope:newly_created_artifact ?= {
		equip_artifact_to_owner_replace = yes
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = tfw_e_winter_court_crown
		save_scope_as = epic
	}
}

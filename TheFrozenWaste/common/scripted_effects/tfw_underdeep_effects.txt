# Run state lives on the activity as variables: underdeep_floor, underdeep_rooms_done,
# underdeep_gold_found, underdeep_boss_odds, underdeep_supplies (adventurers only).

# Adventurers spend real provisions. Landed rulers pay in gold for costly choices instead.
underdeep_supplies_effect = {
	if = {
		limit = { is_landless_adventurer = yes }
		domicile ?= { change_provisions = $AMOUNT$ }
	}
	else = {
		remove_short_term_gold = $GOLD$
	}
}

underdeep_supplies_only_effect = {
	if = {
		limit = { is_landless_adventurer = yes }
		domicile ?= { change_provisions = $AMOUNT$ }
	}
}

# Every find is scaled by underdeep_loot_mult. The Two Cups uses underdeep_jackpot_effect instead.
underdeep_loot_effect = {
	save_temporary_scope_value_as = {
		name = ud_loot
		value = {
			value = $AMOUNT$
			multiply = underdeep_loot_mult
		}
	}
	underdeep_add_found_gold_effect = { AMOUNT = scope:ud_loot }
}

underdeep_jackpot_effect = {
	save_temporary_scope_value_as = { name = ud_loot value = $AMOUNT$ }
	underdeep_add_found_gold_effect = { AMOUNT = scope:ud_loot }
}

underdeep_add_found_gold_effect = {
	add_gold = $AMOUNT$
	hidden_effect = {
		involved_activity = {
			if = {
				limit = { NOT = { has_variable = underdeep_gold_found } }
				set_variable = { name = underdeep_gold_found value = 0 }
			}
			change_variable = { name = underdeep_gold_found add = $AMOUNT$ }
		}
	}
}

# Keeps PERCENT of the gold found on this run and takes back the rest.
underdeep_loot_keep_effect = {
	custom_tooltip = $TT$
	hidden_effect = {
		if = {
			limit = { involved_activity = { has_variable = underdeep_gold_found } }
			save_scope_value_as = {
				name = ud_gold_loss
				value = {
					value = involved_activity.var:underdeep_gold_found
					multiply = { value = 100 subtract = $PERCENT$ divide = 100 }
				}
			}
			remove_short_term_gold = scope:ud_gold_loss
			involved_activity = {
				change_variable = { name = underdeep_gold_found multiply = $PERCENT$ }
				change_variable = { name = underdeep_gold_found divide = 100 }
			}
		}
	}
}

underdeep_wound_effect = {
	if = {
		limit = {
			this = root
			involved_activity = {
				any_attending_character = {
					has_activity_intent = underdeep_watch_intent
					NOT = { this = root }
				}
			}
		}
		random_list = {
			50 = {
				involved_activity = {
					random_attending_character = {
						limit = {
							has_activity_intent = underdeep_watch_intent
							NOT = { this = root }
						}
						increase_wounds_effect = { REASON = fight }
						complete_activity_intent = yes
						add_opinion = { target = root modifier = underdeep_opinion_good }
					}
				}
			}
			50 = {
				increase_wounds_effect = { REASON = fight }
			}
		}
	}
	else = {
		increase_wounds_effect = { REASON = fight }
	}
}

	hidden_effect = {
		if = {
			limit = { underdeep_floor_is = { F = 1 } involved_activity = { has_variable = underdeep_next_$KIND$ } }
			underdeep_enter_picked_room_effect = { KIND = $KIND$ }

			involved_activity = {
				remove_variable = underdeep_next_treasure
				remove_variable = underdeep_next_danger
				remove_variable = underdeep_next_whispers
				remove_variable = underdeep_next_rest

# Grants the rewards for the run, then moves on to any personal ending.
underdeep_finish_effect = {
	save_scope_value_as = { name = ud_xp value = underdeep_run_xp }
	save_scope_value_as = { name = ud_prestige value = underdeep_total_prestige }
	add_prestige = scope:ud_prestige
	if = {
		limit = { NOT = { has_trait_xp = { trait = delver value >= 100 } } }
		hidden_effect = { involved_activity = { set_variable = { name = underdeep_delver_gained value = scope:ud_xp } } }
	}
	involved_activity = { set_variable = { name = underdeep_prestige_won value = scope:ud_prestige } }
	underdeep_intent_finish_effect = yes
	if = {
		limit = { NOT = { has_trait = delver } }
		add_trait = delver
	}
	if = {
		limit = { NOT = { has_trait_xp = { trait = delver value >= 100 } } }
		add_trait_xp = { trait = delver value = scope:ud_xp }
	}
	hidden_effect = {
		if = {
			limit = {
				involved_activity = {
					has_variable = underdeep_doppel
					var:underdeep_doppel = { is_alive = yes }
				}
			}
			involved_activity.var:underdeep_doppel = { save_scope_as = ud_doppel }
			trigger_event = { id = underdeep.1255 days = 2 }
		}
		else = {
			underdeep_finish_route_effect = yes
		}
	}
}

# Scope is the party member who would die. Root is the host of the run.
underdeep_party_death_effect = {
	if = {
		limit = { root = { has_activity_intent = underdeep_home_intent } }
		random_list = {
			50 = {
				increase_wounds_effect = { REASON = fight }
			}
			50 = {
				root = { involved_activity = { change_variable = { name = underdeep_party_deaths add = 1 } } }
				death = { death_reason = death_vanished }
			}
		}
	}
	else = {
		root = { involved_activity = { change_variable = { name = underdeep_party_deaths add = 1 } } }
		death = { death_reason = death_vanished }
	}
}

underdeep_intent_finish_effect = {
	if = {
		limit = { has_activity_intent = underdeep_plunder_intent }
		complete_activity_intent = yes
	}
	else_if = {
		limit = { has_activity_intent = underdeep_renown_intent }
		if = {
			limit = { involved_activity.var:underdeep_floor >= underdeep_max_floor }
			complete_activity_intent = yes
		}
	}
	else_if = {
		limit = { has_activity_intent = underdeep_lore_intent }
		add_learning_lifestyle_xp = 300
		complete_activity_intent = yes
	}
	else_if = {
		limit = { has_activity_intent = underdeep_home_intent }
		if = {
			limit = { involved_activity = { var:underdeep_party_deaths <= 0 } }
			involved_activity = {

# A and B grow closer. Friends if B already liked A or MAJOR, best friends if already friends and MAJOR.
underdeep_bond_effect = {
	$A$ = {
		if = {
			limit = {
				always = $MAJOR$
				has_relation_friend = $B$
				can_set_relation_best_friend_trigger = { CHARACTER = $B$ }
			}
			set_relation_best_friend = { target = $B$ reason = underdeep_relation_bond }
		}
		else_if = {
			limit = {
				can_set_relation_friend_trigger = { CHARACTER = $B$ }
				OR = {
					always = $MAJOR$
					$B$ = { opinion = { target = $A$ value >= 30 } }
				}
			}
			set_relation_friend = { target = $B$ reason = underdeep_relation_bond }
		}
		else = {
			add_opinion = { target = $B$ modifier = underdeep_opinion_good }
			$B$ = { add_opinion = { target = $A$ modifier = underdeep_opinion_good } }
		}
	}
}

# A and B fall out. Rivals if B already disliked A or MAJOR, nemeses if already rivals and MAJOR.
underdeep_feud_effect = {
	$A$ = {
		if = {
			limit = {
				always = $MAJOR$
				has_relation_rival = $B$
				can_set_relation_nemesis_trigger = { CHARACTER = $B$ }
			}
			set_relation_nemesis = { target = $B$ reason = underdeep_relation_feud }
		}
		else_if = {
			limit = {
				can_set_relation_rival_trigger = { CHARACTER = $B$ }
				OR = {
					always = $MAJOR$
					$B$ = { opinion = { target = $A$ value <= -30 } }
				}
			}
			set_relation_rival = { target = $B$ reason = underdeep_relation_feud }
		}
		else = {
			add_opinion = { target = $B$ modifier = underdeep_opinion_bad }
			$B$ = { add_opinion = { target = $A$ modifier = underdeep_opinion_bad } }
		}
	}
}

# An old rivalry is laid down. If there was none, they simply think better of each other.
underdeep_end_feud_effect = {
	$A$ = {
		if = {
			limit = { has_relation_rival = $B$ }
			remove_relation_rival = $B$
		}
		else = {
			add_opinion = { target = $B$ modifier = underdeep_opinion_good }
			$B$ = { add_opinion = { target = $A$ modifier = underdeep_opinion_good } }
		}
	}
}

	custom_tooltip = underdeep_tt_single_combat_$BOSS$
	save_scope_value_as = { name = ud_duel_win value = event_id:underdeep.$WIN$ }

	save_scope_value_as = { name = ud_boss_prowess value = underdeep_duel_value_$BOSS$ }
	hidden_effect = {
		underdeep_boss_duelist_$BOSS$_effect = yes
		scope:ud_boss = { add_prowess_skill = scope:ud_boss_prowess }
		# The duel is set up in its own event: tooltips would otherwise check it before the boss exists.
		trigger_event = underdeep.6403
	}
}

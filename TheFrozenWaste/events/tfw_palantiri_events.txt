tfw_palantiri.02 = { # palintir fishing
	content_source = realms_dlc
	type = character_event
	title = tfw_palantiri.02.t
	desc = tfw_palantiri.02.desc
	theme = sailing_activity
	
	override_background = {
		reference = wilderness_arctic
	}
	
	left_portrait = {
		character = root
		animation = happiness
	}
	artifact = { # To display the artifact in the event-window
		target = scope:artifact1
		position = lower_center_portrait
	}
	artifact = { # To display the artifact in the event-window
		target = scope:artifact2
		position = lower_right_portrait
	}
	
	immediate = {
		# Mark the original Arnor stones as recovered across all three mods.
		set_global_variable = { name = tfw_palantiri_recovered value = yes }
		create_artifact_palantir_annuminas_effect = { OWNER = this }
		scope:newly_created_artifact = {
			save_scope_as = artifact1
		}
		create_artifact_palantir_amonsul_effect = { OWNER = this }
		scope:newly_created_artifact = {
			save_scope_as = artifact2
		}
	}

	option = {
		name = tfw_palantiri.02.a
		
		add_prestige = 2000
		dynasty = { add_dynasty_prestige = 1000 }
		

	}
}

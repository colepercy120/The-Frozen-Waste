#############################################

tfw_story.003 = {
	type = character_event
	title = tfw_story.003.t
	desc = tfw_story.003.desc
	theme = war
	
	override_background = {
		reference = wilderness_underground
	}

	left_portrait = {
		character = root
		animation = admiration
	}

	# Once the actual artifact creation effect exists,
	# the newly-created sword can be displayed in this event.
	#
	artifact = {
		target = scope:faron_sword
		position = lower_center_portrait
	}
	
	immediate = {
		create_artifact_maegdaes_effect = { OWNER = this }
		scope:newly_created_artifact = {
			save_scope_as = faron_sword
		}
	}

	option = {
		name = tfw_story.003.a

		
		trigger_event = {
			id = tfw_story.004
		}
	}
}

#############################################

tfw_story.004 = {
	type = character_event
	title = tfw_story.004.t
	desc = tfw_story.004.desc
	theme = war
	
	override_background = {
		reference = dragon_generic
	}

	left_portrait = {
		character = root
		animation = aggressive_sword
	}

	# Faron is an ancient female cold-drake who has inhabited
	# Tol Faron for centuries. If she is represented as a character,
	# add her portrait here and her death effect to tfw_story.006.
	#
	# right_portrait = {
	#	character = character:TEMP_FARON
	#	animation = threatening
	# }


	###############################
	# MARTIAL + PROWESS
	###############################

	option = {
		name = tfw_story.004.a

		show_as_tooltip = {
			duel = {
				skills = { prowess martial }
	
				# TEMP:
				# Increase this number to make the challenge harder.
				value = 30
	
				25 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 1
					}
	
					send_interface_toast = {
						title = tfw_story.004.a.success
					}
	
					custom_tooltip = {
						text = tfw_story.004.a.success
					}
	
					trigger_event = {
						id = tfw_story.006
					}
				}
	
				75 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -1
					}
	
					send_interface_toast = {
						title = tfw_story.004.a.failure
					}
	
					death = {
						death_reason = death_duel
					}
				}
			}
		}
		trigger_event = {
			id = tfw_story.006
		}

		ai_chance = {
			base = 100

			
		}
	}


	###############################
	# INTRIGUE + PROWESS
	###############################

		name = tfw_story.004.b

		}
